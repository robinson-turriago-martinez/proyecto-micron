    ROL {
        int id_rol PK
        varchar nombre_rol UK
        varchar descripcion
    }
    ZONA {
        int id_zona PK
        varchar nombre_zona
        varchar descripcion
    }
    TIPO_RESIDUO {
        int id_tipo_residuo PK
        varchar nombre UK
        varchar descripcion
    }
    USUARIO {
        int id_usuario PK
        varchar nombres
        varchar apellidos
        varchar cedula UK
        varchar correo UK
        varchar contrasena
        varchar telefono
        timestamp fecha_registro
        int id_rol FK
        int id_zona FK
        varchar estado
    }
    UBICACION {
        int id_ubicacion PK
        varchar direccion
        varchar barrio
        decimal latitud
        decimal longitud
        int id_zona FK
    }
    REPORTE {
        int id_reporte PK
        timestamp fecha_creacion
        text descripcion
        varchar estado
        varchar prioridad
        varchar nivel_afectacion
        timestamp fecha_actualizacion
        int id_usuario FK
        int id_ubicacion FK
        int id_tipo_residuo FK
    }
    EVIDENCIA {
        int id_evidencia PK
        varchar tipo_archivo
        varchar url_archivo
        timestamp fecha_carga
        int id_reporte FK
    }
    ASIGNACION {
        int id_asignacion PK
        timestamp fecha_asignacion
        text comentario
        int id_reporte FK
        int id_operario FK
        int id_supervisor FK
    }
    SEGUIMIENTO {
        int id_seguimiento PK
        timestamp fecha
        varchar estado_anterior
        varchar estado_nuevo
        text comentario
        int id_reporte FK
        int id_usuario FK
    }
    NOTIFICACION {
        int id_notificacion PK
        varchar mensaje
        timestamp fecha_envio
        boolean leida
        int id_usuario FK
        int id_reporte FK
    }
